  container: 'container',
  autoFit: true,
  height: 720,
  padding: 50,
});

chart.options({
  type: 'interval',
  coordinate: { type: 'polar' },
  data: data,
  encode: { x: 'direction', y: 'value', color: 'level', size: 18 },
  transform: [{ type: 'stackY' }],
  tooltip: {
    title: (d) => d.direction,
    items: [
      (d, i, data, column) => ({
        name: d.level,
        value: d.value,
        channel: 'y',
      }),
    ],
  },
  interaction: {
    tooltip: {
      shared: true,
    },
  },
  scale: { color: { range: colors } },
  axis: {
    x: {
      line: true,
      grid: true,
      gridLineDash: [0, 0],
      gridLineWidth: 1,
    },
    y: {
      title: false,
      line: true,
      gridLineWidth: 1,
    },
  },
});

chart.render();
